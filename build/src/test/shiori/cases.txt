# Cases for the shiori suite.
#
#   GET <ID> [<ref0>,<ref1>,...]      SHIORI/3.0 GET request
#   NOTIFY <ID> [<ref0>,<ref1>,...]   SHIORI/3.0 NOTIFY request
#   REQ <first line> [| <Header>: <value> ...]   a request as it is ("Charset: Shift_JIS", "Sender: SSP" and
#                                     "SecurityLevel: local" are added unless the case gives its own)
#   ~ <text>                          the response must contain <text> (a substring)
#
# All the requests of this file are sent in order to one loaded module, so a
# case can depend on the cases before it (see ownerghostname below).
# Requests are sent with "Sender: SSP" and "SecurityLevel: local".
# Use ASCII only.

# event answered from the dictionary (backslashes are passed through)
GET OnBoot
~ SHIORI/3.0 200 OK
~ Value: \0\s[0]Hello\e
~ Charset: Shift_JIS

# References reach System.Request.Reference<n>
GET OnRef a,b
~ SHIORI/3.0 200 OK
~ Value: r0=a r1=b

# System.Sender.Path holds the sender path of the request (SecurityLevel: local)
GET OnSenderPath
~ Value: local

# resource from the dictionary
GET homeurl
~ SHIORI/3.0 200 OK
~ Value: http://example.invalid/

# resources the module answers by itself
GET craftman
~ SHIORI/3.0 200 OK
~ Value: KawariDeveloperTeam

GET shioriid
~ SHIORI/3.0 200 OK
~ Value: KAWARI.kdt

# nothing to say: 204
GET OnNoSuchEvent
~ SHIORI/3.0 204 No Content

# NOTIFY never has a value
NOTIFY OnSomething
~ SHIORI/3.0 204 No Content

# ownerghostname becomes the Sender header of the following responses
NOTIFY ownerghostname testghost
~ SHIORI/3.0 204 No Content

GET OnBoot
~ SHIORI/3.0 200 OK
~ Sender: testghost

# The Charset of a request is answered back, whatever the module can convert (Shift_JIS and UTF-8 only before).
# The name is matched without case and without '-' '_' '.'. An unknown name is read as Shift_JIS.
REQ GET SHIORI/3.0 | ID: OnBoot | Charset: EUC-JP
~ SHIORI/3.0 200 OK
~ Value: \0\s[0]Hello\e
~ Charset: EUC-JP

REQ GET SHIORI/3.0 | ID: OnBoot | Charset: big5
~ Value: \0\s[0]Hello\e
~ Charset: Big5

REQ GET SHIORI/3.0 | ID: OnBoot | Charset: ISO-8859-1
~ Value: \0\s[0]Hello\e
~ Charset: ISO-8859-1

REQ GET SHIORI/3.0 | ID: OnBoot | Charset: utf_8
~ Value: \0\s[0]Hello\e
~ Charset: UTF-8

REQ GET SHIORI/3.0 | ID: OnBoot | Charset: x-no-such-charset
~ Value: \0\s[0]Hello\e
~ Charset: Shift_JIS
