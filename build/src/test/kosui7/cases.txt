# Cases for the kosui7 suite (the Kawari 7 compatible mode).
#
#   echo <input>   input line in echo mode (evaluated as a sentence)
#   cmd <input>    input line in command mode (evaluated as $(<input>))
#   = <text>       expected output (a lone "=" is an empty line)
#
# The folder has kawari.ini and no kawarirc.kis, so kosui runs in the Kawari 7 compatible mode.
# Every case runs in a fresh copy of this folder (without cases.txt).
#
# The expected outputs of the sections before "extensions" were taken from the real Kawari 7.5.0
# (kosui.exe 7.5.0): the same lines gave the same output there. Use ASCII only.

# entry reference
echo ${greet}
= Hello
echo ${nest}
= Hello world!
echo plain text
= plain text
echo ${nosuch}
=
echo ${ b }
= x
echo ${one}${one}
= onlyonly

# entry names are flat strings in 7: "dots" "dots." "dots..x" are three different entries
echo ${dots}
= plain
echo ${dots.}
= dotted
echo ${dots..x}
= doubled
echo ${dots.x}
=

# @ is an ordinary character of an entry name in 7 (not a local entry as in 9)
cmd set @loc 5 ; echo ${@loc}
= 5
cmd set @loc.a 1 ; echo ${@loc.a}/${@loc}
= 1/
echo ${@none}
=

# lexical rules: separators are plain text in a sentence, a quoted text is not expanded
echo a (b) c
= a (b) c
echo a;b:c!d&e|f
= a;b:c!d&e|f
echo a,b
= a,b
echo "he said \"hi\""
= he said "hi"
echo $x $() $
= $x $() $
echo a   b
= a   b
echo ${quote}
= a , b
echo "${greet}"
= ${greet}
echo $(echo a
= a

# history reference counts only the entry calls (${N} is not counted, $() is not counted)
echo ${b}$(echo q)${0}/${1}
= xqx/
echo ${b}${n}${1}/${0}
= x33/x
echo ${b}${n}${0}${1}${2}
= x3x3
echo $(echo ${b} ${n})${0}${1}
= x 3

# if takes a condition, a word and an optional else word (no else keyword)
echo $(if 1 yes no)
= yes
echo $(if "" yes no)
= no
echo $(if 0 yes)
=
echo $(if 0 yes no)
= no
echo $(if false yes no)
= no
echo $(if 1 yes else no)
=
echo $(if 1 $(echo x)$(echo y) z)
= xy
cmd if 1 a b
= a
cmd if x a
= a

# save and load: a word is written in the 7 syntax and read back (the backslash must not double)
cmd set r1 "a\b" ; save r.txt r1 ; clear r1 ; load r.txt ; array r1 0
= a\b
cmd set r1 "a\b" ; save r.txt r1 ; clear r1 ; load r.txt ; load r.txt ; array r1 1
= a\b
cmd set r2 "x,y" ; save r.txt r2 ; clear r2 ; load r.txt ; size r2
= 1
cmd set r3 "say \"hi\"" ; save r.txt r3 ; clear r3 ; load r.txt ; array r3 0
= say hi
cmd set r4 "$(if 1 a b)" ; save r.txt r4 ; clear r4 ; load r.txt ; get r4 0
= $("if" "1" "a" "b" )
cmd set r4 "$(if 1 a b)" ; save r.txt r4 ; clear r4 ; load r.txt ; array r4 0
= a
cmd set r5 "${b}${0}" ; save r.txt r5 ; clear r5 ; load r.txt ; array r5 0
= xx
cmd adddict r6 one ; adddict r6 "t w o" ; save r.txt r6 ; clear r6 ; load r.txt ; enumerate r6

cmd set r7 secret ; savecrypt r.txt r7 ; clear r7 ; load r.txt ; array r7 0

cmd set r8 "a b" ; get r8 0
